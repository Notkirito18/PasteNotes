                Log.Info("PasteNotes starting");

                AppDomain.CurrentDomain.UnhandledException += OnUnhandledException;

                {
                    Application.Run(context);
                }
                return 0;
            }
            catch (DataVersionException ex)

                Log.Error("Data version is newer than this build", ex);
                MessageBox.Show(
                    "The PasteNotes data file was created by a newer version.\r\n\r\n" +
                    AppPaths.DataFile + "\r\n\r\n" +
                    "This build will not modify it. Use a newer PasteNotes build.",
                    "PasteNotes",

                    "PasteNotes",
                    MessageBoxButtons.OK,
                    MessageBoxIcon.Error);
                return 1;
            }
            finally
            {
                if (createdNew)
                {
                    try { mutex.ReleaseMutex(); }
                    catch { }
                }
                mutex.Dispose();
                Log.Info("PasteNotes stopped");
            }
        }

    internal static class Log
    {
        private const long MaxBytes = 1000000;
        private static readonly object Sync = new object();
        private static string _path;

        public static void Initialize(string path)
        {
            try
            {
                _path = path;
                string directory = Path.GetDirectoryName(path);
                if (!string.IsNullOrEmpty(directory)) Directory.CreateDirectory(directory);
            }
            catch
            {
                _path = null;
            }
        }

        public static void Info(string message)
        {
            Write("INFO", message);
        }

        public static void Warn(string message)
        {
            Write("WARN", message);
        }

        public static void Error(string message, Exception exception)
        {
            string text = message;
            if (exception != null)
            {
                text += "\r\n" + exception;
            }
            Write("ERROR", text);
        }

        private static void Write(string level, string message)
        {
            if (string.IsNullOrEmpty(_path)) return;
            lock (Sync)
            {
                try
                {
                    RotateIfNeeded();
                    string line = string.Format(
                        CultureInfo.InvariantCulture,
                        "{0:yyyy-MM-dd HH:mm:ss} {1} {2}\r\n",
                        DateTime.Now,
                        level,
                        message);
                    File.AppendAllText(_path, line, new UTF8Encoding(false));
                }
                catch
                {
                }
            }
        }

        private static void RotateIfNeeded()
        {
            try
            {
                if (!File.Exists(_path)) return;
                FileInfo info = new FileInfo(_path);
                if (info.Length < MaxBytes) return;

                for (int index = 3; index >= 1; index--)
                {
                    string source = _path + "." + index.ToString(CultureInfo.InvariantCulture);
                    string target = _path + "." + (index + 1).ToString(CultureInfo.InvariantCulture);
                    if (File.Exists(source))
                    {
                        if (File.Exists(target)) File.Delete(target);
                        File.Move(source, target);
                    }
                }
                string first = _path + ".1";
                if (File.Exists(first)) File.Delete(first);
                File.Move(_path, first);
            }
            catch
            {
            }
        }
    }

    internal static class RtlHelper
    {
        public static bool IsRtl(string text)
        {
            return DirectionOf(text) == 1;
        }

        private static int DirectionOf(string text)
        {
            if (string.IsNullOrEmpty(text)) return 0;

            for (int index = 0; index < text.Length; index++)
            {
                char c = text[index];
                int code = c;

                if (char.IsHighSurrogate(c) && index + 1 < text.Length && char.IsLowSurrogate(text[index + 1]))
                {
                    int point = char.ConvertToUtf32(c, text[index + 1]);
                    if (point >= 0x10800 && point <= 0x10FFF) return 1;
                    if (point >= 0x1E800 && point <= 0x1EFFF) return 1;
                    index++;
                    continue;
                }

                if (code >= 0x0590 && code <= 0x05FF) return 1;
                if (code >= 0x0600 && code <= 0x06FF)
                {
                    if ((code >= 0x0660 && code <= 0x0669) || (code >= 0x06F0 && code <= 0x06F9))
                    {
                        continue;
                    }
                    return 1;
                }
                if (code >= 0x0750 && code <= 0x077F) return 1;
                if (code >= 0x08A0 && code <= 0x08FF) return 1;
                if (code >= 0xFB1D && code <= 0xFDFF) return 1;
                if (code >= 0xFE70 && code <= 0xFEFF) return 1;

                if ((code >= 'A' && code <= 'Z') || (code >= 'a' && code <= 'z')) return -1;
                if (char.IsLetter(c)) return -1;
            }
            return 0;
        }

        public static void Apply(TextBox box)
        {
            if (box == null) return;
            ApplyDirection(box, IsRtl(box.Text));
        }

        public static void ApplyEditor(TextBox box)
        {
            if (box == null) return;
            string text = box.Text;
            if (string.IsNullOrEmpty(text))
            {
                ApplyDirection(box, false);
                return;
            }

            string normalized = text.Replace("\r\n", "\n").Replace('\r', '\n');
            int caret = box.SelectionStart;
            if (caret < 0) caret = 0;
            if (caret > normalized.Length) caret = normalized.Length;

            int lineIndex = 0;
            for (int index = 0; index < caret; index++)
            {
                if (normalized[index] == '\n') lineIndex++;
            }

            string[] lines = normalized.Split('\n');
            int direction = 0;

            if (lineIndex < lines.Length) direction = DirectionOf(lines[lineIndex]);
            for (int index = lineIndex - 1; index >= 0 && direction == 0; index--)
            {
                direction = DirectionOf(lines[index]);
            }
            for (int index = 0; index < lines.Length && direction == 0; index++)
            {
                direction = DirectionOf(lines[index]);
            }

            ApplyDirection(box, direction == 1);
        }

        private static void ApplyDirection(TextBox box, bool rtl)
        {
            RightToLeft desired = rtl ? RightToLeft.Yes : RightToLeft.No;
            HorizontalAlignment alignment = rtl ? HorizontalAlignment.Right : HorizontalAlignment.Left;
            if (box.RightToLeft != desired)
            {
                box.RightToLeft = desired;
            }
            if (box.TextAlign != alignment)
            {
                box.TextAlign = alignment;
            }
        }
    }

        public static bool ForceForeground(IntPtr hWnd)
        {
            if (hWnd == IntPtr.Zero || !IsWindow(hWnd)) return false;
            if (GetForegroundWindow() == hWnd) return true;

            if (IsIconic(hWnd)) ShowWindow(hWnd, SW_RESTORE);

            IntPtr foreground = GetForegroundWindow();
            uint currentThread = GetCurrentThreadId();
            uint foregroundThread = foreground == IntPtr.Zero ? 0 : GetWindowThreadProcessId(foreground, IntPtr.Zero);
            uint targetThread = GetWindowThreadProcessId(hWnd, IntPtr.Zero);
            bool attachedForeground = false;
            bool attachedTarget = false;

            try
            {
                if (foregroundThread != 0 && foregroundThread != currentThread)
                {
                    attachedForeground = AttachThreadInput(currentThread, foregroundThread, true);
                }
                if (targetThread != 0 && targetThread != currentThread && targetThread != foregroundThread)
                {
                    attachedTarget = AttachThreadInput(currentThread, targetThread, true);
                }

                SetForegroundWindow(hWnd);
                BringWindowToTop(hWnd);
                SetFocus(hWnd);
            }
            finally
            {
                if (attachedTarget) AttachThreadInput(currentThread, targetThread, false);
                if (attachedForeground) AttachThreadInput(currentThread, foregroundThread, false);
            }

            if (GetForegroundWindow() != hWnd)
            {
                try { SwitchToThisWindow(hWnd, true); }
                catch { }
            }
            return GetForegroundWindow() == hWnd;
        }

        public static bool SendPaste()
        {
            INPUT[] inputs = new INPUT[4];
            inputs[0] = KeyInput((ushort)VK_CONTROL, false);
            inputs[1] = KeyInput((ushort)VK_V, false);
            inputs[2] = KeyInput((ushort)VK_V, true);
            inputs[3] = KeyInput((ushort)VK_CONTROL, true);
            uint sent = SendInput((uint)inputs.Length, inputs, Marshal.SizeOf(typeof(INPUT)));
            return sent == (uint)inputs.Length;
        }

        private static INPUT KeyInput(ushort virtualKey, bool keyUp)
        {
            INPUT input = new INPUT();
            input.type = INPUT_KEYBOARD;
            input.U.ki.wVk = virtualKey;
            input.U.ki.wScan = 0;
            input.U.ki.dwFlags = keyUp ? KEYEVENTF_KEYUP : 0;
            input.U.ki.time = 0;
            input.U.ki.dwExtraInfo = UIntPtr.Zero;
            return input;
        }

    internal static class IconFactory
    {
        private static Icon _applicationIcon;

        public static Icon ApplicationIcon
        {
            get
            {
                if (_applicationIcon == null) _applicationIcon = Create(false);
                return _applicationIcon;
            }
        }

        public static Icon Create(bool paused)
        {
            using (Bitmap bitmap = new Bitmap(32, 32))
            using (Graphics graphics = Graphics.FromImage(bitmap))
            {
                graphics.SmoothingMode = SmoothingMode.AntiAlias;
                graphics.Clear(Color.Transparent);

                Color noteColor = paused ? Color.FromArgb(154, 160, 170) : Color.FromArgb(243, 193, 75);
                Color clipColor = Color.FromArgb(42, 47, 58);
                Color lineColor = Color.FromArgb(77, 58, 18);

                using (GraphicsPath note = UiHelpers.RoundRect(new Rectangle(6, 5, 20, 23), 4))
                using (SolidBrush brush = new SolidBrush(noteColor))
                {
                    graphics.FillPath(brush, note);
                }
                using (GraphicsPath clip = UiHelpers.RoundRect(new Rectangle(11, 3, 10, 6), 2))
                using (SolidBrush brush = new SolidBrush(clipColor))
                {
                    graphics.FillPath(brush, clip);
                }
                using (Pen pen = new Pen(lineColor, 1.6f))
                {
                    graphics.DrawLine(pen, 10, 15, 22, 15);
                    graphics.DrawLine(pen, 10, 19, 22, 19);
                    graphics.DrawLine(pen, 10, 23, 18, 23);
                }

                if (paused)
                {
                    using (SolidBrush brush = new SolidBrush(Color.FromArgb(30, 35, 48)))
                    {
                        graphics.FillEllipse(brush, 18, 18, 12, 12);
                    }
                    using (SolidBrush brush = new SolidBrush(Color.FromArgb(243, 193, 75)))
                    {
                        graphics.FillRectangle(brush, 21, 21, 2, 6);
                        graphics.FillRectangle(brush, 25, 21, 2, 6);
                    }
                }

                IntPtr handle = bitmap.GetHicon();
                try
                {
                    using (Icon temporary = Icon.FromHandle(handle))
                    {
                        return (Icon)temporary.Clone();
                    }
                }
                finally
                {
                    NativeMethods.DestroyIcon(handle);
                }
            }
        }
    }

    internal static class UiHelpers
    {
        public const int EM_SETCUEBANNER = 0x1501;

        public static GraphicsPath RoundRect(Rectangle bounds, int radius)
        {
            GraphicsPath path = new GraphicsPath();
            int diameter = Math.Max(1, radius * 2);
            if (diameter > bounds.Width) diameter = bounds.Width;
            if (diameter > bounds.Height) diameter = bounds.Height;
            Rectangle arc = new Rectangle(bounds.Location, new Size(diameter, diameter));
            path.AddArc(arc, 180, 90);
            arc.X = bounds.Right - diameter;
            path.AddArc(arc, 270, 90);
            arc.Y = bounds.Bottom - diameter;
            path.AddArc(arc, 0, 90);
            arc.X = bounds.Left;
            path.AddArc(arc, 90, 90);
            path.CloseFigure();
            return path;
        }

        public static void SetCueBanner(TextBox box, string text)
        {
            if (box == null || box.IsHandleCreated == false) return;
            NativeMethods.SendMessage(box.Handle, EM_SETCUEBANNER, new IntPtr(1), text);
        }

        public static bool IsShellWindow(IntPtr hWnd)
        {
            string className = NativeMethods.GetWindowClass(hWnd);
            return string.Equals(className, "Progman", StringComparison.OrdinalIgnoreCase)
                || string.Equals(className, "WorkerW", StringComparison.OrdinalIgnoreCase)
                || string.Equals(className, "Shell_TrayWnd", StringComparison.OrdinalIgnoreCase);
        }

        public static int MeasureListHeight(Font font, int padding)
        {
            return Math.Max(18, TextRenderer.MeasureText("Ag", font).Height + padding);
        }

        public List<TemplateData> Templates;

        public GroupData()
        {
            Id = DataStore.NewId();
            Name = "New group";
            Hotkey = "";
            Color = Palette.DefaultKey;
            Templates = new List<TemplateData>();
        }
    }

    internal sealed class AppData
    {
        public int Version;
        public Dictionary<string, object> Settings;
        public List<GroupData> Groups;

        public static AppData CreateEmpty()
        {
            AppData data = new AppData();
            data.Version = DataStore.DataVersion;
            data.Settings = DataStore.DefaultSettings();
            data.Groups = new List<GroupData>();
            return data;
        }
    }

    internal sealed class DataStore
    {
        public const int DataVersion = 1;

        public string FilePath { get; private set; }
        public AppData Data { get; private set; }
        public bool LastSaveFailed { get; private set; }

        public DataStore(string path)
        {
            FilePath = path;
            Data = AppData.CreateEmpty();
        }

        public static Dictionary<string, object> DefaultSettings()
        {
            Dictionary<string, object> settings = new Dictionary<string, object>();
            settings["restore_clipboard"] = true;
            return settings;
        }

        public static string NewId()
        {
            return Guid.NewGuid().ToString("N");
        }

        public void Load()
        {
            if (!File.Exists(FilePath))
            {
                Seed();
                Save();
                return;
            }

            AppData loaded = null;
            try
            {
                loaded = JsonCodec.ParseAppData(ReadText(FilePath));
            }
            catch (DataVersionException)
            {
                throw;
            }
            catch (Exception ex)
            {
                Log.Warn("Could not read data file: " + ex.Message);
            }

            if (loaded != null)
            {
                Data = loaded;
                Normalize();
                return;
            }

            string temporary = FilePath + ".tmp";
            if (File.Exists(temporary))
            {
                try
                {
                    Data = JsonCodec.ParseAppData(ReadText(temporary));
                    Normalize();
                    Save();
                    Log.Info("Recovered data from data.json.tmp");
                    return;
                }
                catch (DataVersionException)
                {
                    throw;
                }
                catch (Exception ex)
                {
                    Log.Warn("Could not recover data.json.tmp: " + ex.Message);
                }
            }

            string backup = FilePath + ".bak";
            if (File.Exists(backup))
            {
                try
                {
                    Data = JsonCodec.ParseAppData(ReadText(backup));
                    Normalize();
                    Save();
                    Log.Info("Recovered data from data.json.bak");
                    return;
                }
                catch (DataVersionException)
                {
                    throw;
                }
                catch (Exception ex)
                {
                    Log.Warn("Could not recover data.json.bak: " + ex.Message);
                }
            }

            QuarantineCorruptFile();
            Seed();
            Save();
        }

        public void Save()
        {
            try
            {
                string directory = Path.GetDirectoryName(FilePath);
                if (!string.IsNullOrEmpty(directory)) Directory.CreateDirectory(directory);

                string temporary = FilePath + ".tmp";
                string backup = FilePath + ".bak";
                string json = JsonCodec.SerializeAppData(Data);
                File.WriteAllText(temporary, json, new UTF8Encoding(false));

                if (File.Exists(FilePath))
                {
                    try
                    {
                        File.Replace(temporary, FilePath, backup, true);
                    }
                    catch
                    {
                        try
                        {
                            File.Copy(FilePath, backup, true);
                            File.Copy(temporary, FilePath, true);
                            File.Delete(temporary);
                        }
                        catch (Exception ex)
                        {
                            LastSaveFailed = true;
                            throw new IOException("Could not replace data.json", ex);
                        }
                    }
                }
                else
                {
                    File.Move(temporary, FilePath);
                }

                LastSaveFailed = false;
            }
            catch
            {
                LastSaveFailed = true;
                throw;
            }
        }

        public void Seed()
        {
            Data = Defaults.CreateAppData();
            Normalize();
        }

        public bool GetBoolSetting(string key, bool fallback)
        {
            if (Data == null || Data.Settings == null) return fallback;
            object value;
            if (!Data.Settings.TryGetValue(key, out value)) return fallback;
            if (value is bool) return (bool)value;
            try { return Convert.ToBoolean(value, CultureInfo.InvariantCulture); }
            catch { return fallback; }
        }

        public string GetStringSetting(string key, string fallback)
        {
            if (Data == null || Data.Settings == null) return fallback;
            object value;
            if (!Data.Settings.TryGetValue(key, out value) || value == null) return fallback;
            return Convert.ToString(value, CultureInfo.InvariantCulture);
        }

        public int GetIntSetting(string key, int fallback)
        {
            if (Data == null || Data.Settings == null) return fallback;
            object value;
            if (!Data.Settings.TryGetValue(key, out value) || value == null) return fallback;
            try { return Convert.ToInt32(value, CultureInfo.InvariantCulture); }
            catch { return fallback; }
        }

        public void SetSetting(string key, object value)
        {
            if (Data.Settings == null) Data.Settings = DefaultSettings();
            Data.Settings[key] = value;
        }

        public GroupData FindGroup(string groupId)
        {
            foreach (GroupData group in Data.Groups)
            {
                if (group.Id == groupId) return group;
            }
            return null;
        }

        public TemplateData FindTemplate(string groupId, string templateId)
        {
            GroupData group = FindGroup(groupId);
            if (group == null) return null;
            foreach (TemplateData template in group.Templates)
            {
                if (template.Id == templateId) return template;
            }
            return null;
        }

        public string UniqueGroupName(string baseName)
        {
            if (string.IsNullOrEmpty(baseName)) baseName = "New group";
            HashSet<string> names = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            foreach (GroupData group in Data.Groups) names.Add(group.Name);
            if (!names.Contains(baseName)) return baseName;
            int index = 2;
            while (names.Contains(baseName + " " + index.ToString(CultureInfo.InvariantCulture))) index++;
            return baseName + " " + index.ToString(CultureInfo.InvariantCulture);
        }

        public string UniqueTemplateTitle(GroupData group, string baseTitle)
        {
            if (string.IsNullOrEmpty(baseTitle)) baseTitle = "New template";
            HashSet<string> titles = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            foreach (TemplateData template in group.Templates) titles.Add(template.Title);
            if (!titles.Contains(baseTitle)) return baseTitle;
            int index = 2;
            while (titles.Contains(baseTitle + " " + index.ToString(CultureInfo.InvariantCulture))) index++;
            return baseTitle + " " + index.ToString(CultureInfo.InvariantCulture);
        }

        public GroupData AddGroup(string name)
        {
            GroupData group = new GroupData();
            group.Name = string.IsNullOrEmpty(name) ? UniqueGroupName("New group") : name.Trim();
            Data.Groups.Add(group);
            return group;
        }

        public bool RemoveGroup(string groupId)
        {
            int before = Data.Groups.Count;
            Data.Groups.RemoveAll(delegate(GroupData group) { return group.Id == groupId; });
            return Data.Groups.Count != before;
        }

        public bool MoveGroup(string groupId, int delta)
        {
            int index = Data.Groups.FindIndex(delegate(GroupData group) { return group.Id == groupId; });
            int target = index + delta;
            if (index < 0 || target < 0 || target >= Data.Groups.Count) return false;
            GroupData item = Data.Groups[index];
            Data.Groups[index] = Data.Groups[target];
            Data.Groups[target] = item;
            return true;
        }

        public TemplateData AddTemplate(GroupData group)
        {
            if (group == null) return null;
            TemplateData template = new TemplateData();
            template.Title = UniqueTemplateTitle(group, "New template");
            group.Templates.Add(template);
            return template;
        }

        public bool RemoveTemplate(GroupData group, string templateId)
        {
            if (group == null) return false;
            int before = group.Templates.Count;
            group.Templates.RemoveAll(delegate(TemplateData template) { return template.Id == templateId; });
            return group.Templates.Count != before;
        }

        public TemplateData DuplicateTemplate(GroupData group, TemplateData source)
        {
            if (group == null || source == null) return null;
            TemplateData copy = new TemplateData();
            copy.Title = UniqueTemplateTitle(group, source.Title + " copy");
            copy.Content = source.Content;
            int index = group.Templates.IndexOf(source);
            if (index < 0) index = group.Templates.Count - 1;
            group.Templates.Insert(index + 1, copy);
            return copy;
        }

        public bool MoveTemplate(GroupData group, TemplateData template, int delta)
        {
            if (group == null || template == null) return false;
            int index = group.Templates.IndexOf(template);
            int target = index + delta;
            if (index < 0 || target < 0 || target >= group.Templates.Count) return false;
            group.Templates[index] = group.Templates[target];
            group.Templates[target] = template;
            return true;
        }

        private static string ReadText(string path)
        {
            return File.ReadAllText(path, Encoding.UTF8);
        }

        private void Normalize()
        {
            if (Data.Settings == null) Data.Settings = DefaultSettings();
            if (!Data.Settings.ContainsKey("restore_clipboard")) Data.Settings["restore_clipboard"] = true;

            HashSet<string> groupIds = new HashSet<string>(StringComparer.Ordinal);
            foreach (GroupData group in Data.Groups)
            {
                if (string.IsNullOrEmpty(group.Id) || groupIds.Contains(group.Id))
                {
                    group.Id = NewId();
                }
                groupIds.Add(group.Id);

                if (group.Name == null || group.Name.Trim().Length == 0) group.Name = "Untitled group";
                if (group.Hotkey == null) group.Hotkey = "";
                group.Color = Palette.Normalize(group.Color);
                if (group.Templates == null) group.Templates = new List<TemplateData>();

                HashSet<string> templateIds = new HashSet<string>(StringComparer.Ordinal);
                foreach (TemplateData template in group.Templates)
                {
                    if (string.IsNullOrEmpty(template.Id) || templateIds.Contains(template.Id))
                    {
                        template.Id = NewId();
                    }
                    templateIds.Add(template.Id);
                    if (template.Title == null || template.Title.Trim().Length == 0) template.Title = "Untitled template";
                    if (template.Content == null) template.Content = "";
                }
            }
        }

        private void QuarantineCorruptFile()
        {
            try
            {
                string directory = Path.GetDirectoryName(FilePath);
                string name = Path.GetFileNameWithoutExtension(FilePath);
                string extension = Path.GetExtension(FilePath);
                string stamp = DateTime.Now.ToString("yyyyMMdd-HHmmss", CultureInfo.InvariantCulture);
                string target = Path.Combine(directory, name + ".corrupt-" + stamp + extension);
                File.Move(FilePath, target);
                Log.Warn("Quarantined unreadable data file as " + target);
            }
            catch (Exception ex)
            {
                Log.Warn("Could not quarantine the data file: " + ex.Message);
            }
        }
    }

    internal static class JsonCodec
    {
        public static AppData ParseAppData(string json)
        {
            JavaScriptSerializer serializer = new JavaScriptSerializer();
            serializer.MaxJsonLength = int.MaxValue;
            serializer.RecursionLimit = 200;

            object raw = serializer.DeserializeObject(json);
            Dictionary<string, object> root = raw as Dictionary<string, object>;
            if (root == null) throw new FormatException("The data file root is not a JSON object.");

            int version = GetInt(root, "version", 1);
            if (version > DataStore.DataVersion) throw new DataVersionException(version);

            AppData data = AppData.CreateEmpty();
            data.Version = DataStore.DataVersion;

            Dictionary<string, object> settings = GetDictionary(root, "settings");
            if (settings != null)
            {
                foreach (KeyValuePair<string, object> pair in settings)
                {
                    data.Settings[pair.Key] = pair.Value;
                }
            }

            object groupsValue;
            if (root.TryGetValue("groups", out groupsValue))
            {
                foreach (object item in AsEnumerable(groupsValue))
                {
                    Dictionary<string, object> groupDictionary = item as Dictionary<string, object>;
                    if (groupDictionary == null) continue;

                    GroupData group = new GroupData();
                    group.Id = GetString(groupDictionary, "id");
                    if (string.IsNullOrEmpty(group.Id)) group.Id = DataStore.NewId();
                    group.Name = GetString(groupDictionary, "name");
                    if (group.Name == null) group.Name = "Untitled group";
                    group.Hotkey = GetString(groupDictionary, "hotkey");
                    if (group.Hotkey == null) group.Hotkey = "";
                    group.Color = Palette.Normalize(GetString(groupDictionary, "color"));

                    object templatesValue;
                    if (groupDictionary.TryGetValue("templates", out templatesValue))
                    {
                        foreach (object templateItem in AsEnumerable(templatesValue))
                        {
                            Dictionary<string, object> templateDictionary = templateItem as Dictionary<string, object>;
                            if (templateDictionary == null) continue;

                            TemplateData template = new TemplateData();
                            template.Id = GetString(templateDictionary, "id");
                            if (string.IsNullOrEmpty(template.Id)) template.Id = DataStore.NewId();
                            template.Title = GetString(templateDictionary, "title");
                            if (template.Title == null) template.Title = "Untitled template";
                            template.Content = GetString(templateDictionary, "content");
                            if (template.Content == null) template.Content = "";
                            group.Templates.Add(template);
                        }
                    }
                    data.Groups.Add(group);
                }
            }

            return data;
        }

        public static string SerializeAppData(AppData data)
        {
            StringBuilder builder = new StringBuilder(65536);
            builder.Append("{\r\n");
            Indent(builder, 1);
            builder.Append("\"version\": ").Append(data.Version.ToString(CultureInfo.InvariantCulture)).Append(",\r\n");
            Indent(builder, 1);
            builder.Append("\"settings\": ");
            WriteValue(builder, data.Settings, 1);
            builder.Append(",\r\n");
            Indent(builder, 1);
            builder.Append("\"groups\": ");
            WriteGroups(builder, data.Groups, 1);
            builder.Append("\r\n}");
            return builder.ToString();
        }

        private static void WriteGroups(StringBuilder builder, List<GroupData> groups, int level)
        {
            if (groups == null || groups.Count == 0)
            {
                builder.Append("[]");
                return;
            }
            builder.Append("[\r\n");
            for (int index = 0; index < groups.Count; index++)
            {
                Indent(builder, level + 1);
                WriteGroup(builder, groups[index], level + 1);
                if (index < groups.Count - 1) builder.Append(',');
                builder.Append("\r\n");
            }
            Indent(builder, level);
            builder.Append(']');
        }

        private static void WriteGroup(StringBuilder builder, GroupData group, int level)
        {
            builder.Append("{\r\n");
            Indent(builder, level + 1);
            builder.Append("\"id\": ");
            WriteString(builder, group.Id);
            builder.Append(",\r\n");
            Indent(builder, level + 1);
            builder.Append("\"name\": ");
            WriteString(builder, group.Name);
            builder.Append(",\r\n");
            Indent(builder, level + 1);
            builder.Append("\"hotkey\": ");
            WriteString(builder, group.Hotkey);
            builder.Append(",\r\n");
            Indent(builder, level + 1);
            builder.Append("\"color\": ");
            WriteString(builder, group.Color);
            builder.Append(",\r\n");
            Indent(builder, level + 1);
            builder.Append("\"templates\": ");
            WriteTemplates(builder, group.Templates, level + 1);
            builder.Append("\r\n");
            Indent(builder, level);
            builder.Append('}');
        }

        private static void WriteTemplates(StringBuilder builder, List<TemplateData> templates, int level)
        {
            if (templates == null || templates.Count == 0)
            {
                builder.Append("[]");
                return;
            }
            builder.Append("[\r\n");
            for (int index = 0; index < templates.Count; index++)
            {
                TemplateData template = templates[index];
                Indent(builder, level + 1);
                builder.Append("{\r\n");
                Indent(builder, level + 2);
                builder.Append("\"id\": ");
                WriteString(builder, template.Id);
                builder.Append(",\r\n");
                Indent(builder, level + 2);
                builder.Append("\"title\": ");
                WriteString(builder, template.Title);
                builder.Append(",\r\n");
                Indent(builder, level + 2);
                builder.Append("\"content\": ");
                WriteString(builder, template.Content);
                builder.Append("\r\n");
                Indent(builder, level + 1);
                builder.Append('}');
                if (index < templates.Count - 1) builder.Append(',');
                builder.Append("\r\n");
            }
            Indent(builder, level);
            builder.Append(']');
        }

        private static void WriteValue(StringBuilder builder, object value, int level)
        {
            if (value == null)
            {
                builder.Append("null");
                return;
            }

            string text = value as string;
            if (text != null)
            {
                WriteString(builder, text);
                return;
            }

            if (value is bool)
            {
                builder.Append((bool)value ? "true" : "false");
                return;
            }

            if (value is int || value is long || value is short || value is byte ||
                value is uint || value is ulong || value is ushort ||
                value is double || value is float || value is decimal)
            {
                builder.Append(Convert.ToString(value, CultureInfo.InvariantCulture));
                return;
            }

            IDictionary<string, object> dictionary = value as IDictionary<string, object>;
            if (dictionary != null)
            {
                if (dictionary.Count == 0)
                {
                    builder.Append("{}");
                    return;
                }
                builder.Append("{\r\n");
                int index = 0;
                foreach (KeyValuePair<string, object> pair in dictionary)
                {
                    Indent(builder, level + 1);
                    WriteString(builder, pair.Key);
                    builder.Append(": ");
                    WriteValue(builder, pair.Value, level + 1);
                    if (index < dictionary.Count - 1) builder.Append(',');
                    builder.Append("\r\n");
                    index++;
                }
                Indent(builder, level);
                builder.Append('}');
                return;
            }

            IEnumerable sequence = value as IEnumerable;
            if (sequence != null)
            {
                List<object> items = new List<object>();
                foreach (object item in sequence) items.Add(item);
                if (items.Count == 0)
                {
                    builder.Append("[]");
                    return;
                }
                builder.Append("[\r\n");
                for (int index = 0; index < items.Count; index++)
                {
                    Indent(builder, level + 1);
                    WriteValue(builder, items[index], level + 1);
                    if (index < items.Count - 1) builder.Append(',');
                    builder.Append("\r\n");
                }
                Indent(builder, level);
                builder.Append(']');
                return;
            }

            WriteString(builder, Convert.ToString(value, CultureInfo.InvariantCulture));
        }

        private static void WriteString(StringBuilder builder, string value)
        {
            builder.Append('"');
            if (value != null)
            {
                for (int index = 0; index < value.Length; index++)
                {
                    char c = value[index];
                    switch (c)
                    {
                        case '"': builder.Append("\\\""); break;
                        case '\\': builder.Append("\\\\"); break;
                        case '\b': builder.Append("\\b"); break;
                        case '\f': builder.Append("\\f"); break;
                        case '\n': builder.Append("\\n"); break;
                        case '\r': builder.Append("\\r"); break;
                        case '\t': builder.Append("\\t"); break;
                        default:
                            if (c < 0x20)
                            {
                                builder.Append("\\u");
                                builder.Append(((int)c).ToString("x4", CultureInfo.InvariantCulture));
                            }
                            else
                            {
                                builder.Append(c);
                            }
                            break;
                    }
                }
            }
            builder.Append('"');
        }

        private static void Indent(StringBuilder builder, int level)
        {
            builder.Append(' ', level * 2);
        }

        private static Dictionary<string, object> GetDictionary(Dictionary<string, object> dictionary, string key)
        {
            object value;
            if (dictionary == null || !dictionary.TryGetValue(key, out value)) return null;
            return value as Dictionary<string, object>;
        }

        private static string GetString(Dictionary<string, object> dictionary, string key)
        {
            object value;
            if (dictionary == null || !dictionary.TryGetValue(key, out value) || value == null) return null;
            return Convert.ToString(value, CultureInfo.InvariantCulture);
        }

        private static int GetInt(Dictionary<string, object> dictionary, string key, int fallback)
        {
            object value;
            if (dictionary == null || !dictionary.TryGetValue(key, out value) || value == null) return fallback;
            try { return Convert.ToInt32(value, CultureInfo.InvariantCulture); }
            catch { return fallback; }
        }

        private static IEnumerable AsEnumerable(object value)
        {
            IEnumerable sequence = value as IEnumerable;
            if (sequence == null || value is string) return new object[0];
            return sequence;
        }
    }

        public static HotkeySpec Parse(string text)
        {
            if (string.IsNullOrEmpty(text) || text.Trim().Length == 0)
            {
                throw new HotkeyException("No hotkey set");
            }

            string work = text.Trim();
            string explicitKey = null;
            if (work.EndsWith("+", StringComparison.Ordinal))
            {
                explicitKey = "+";
                work = work.Substring(0, work.Length - 1);
                if (work.EndsWith("+", StringComparison.Ordinal))
                {
                    work = work.Substring(0, work.Length - 1);
                }
            }

            string[] parts = work.Split(new char[] { '+' }, StringSplitOptions.RemoveEmptyEntries);
            uint modifiers = 0;
            string keyName = explicitKey;

            foreach (string partValue in parts)
            {
                string part = partValue.Trim();
                if (part.Length == 0) continue;
                string upper = part.ToUpperInvariant();

                if (upper == "CTRL" || upper == "CONTROL")
                {
                    modifiers |= NativeMethods.MOD_CONTROL;
                    continue;
                }
                if (upper == "ALT")
                {
                    modifiers |= NativeMethods.MOD_ALT;
                    continue;
                }
                if (upper == "SHIFT")
                {
                    modifiers |= NativeMethods.MOD_SHIFT;
                    continue;
                }
                if (upper == "WIN" || upper == "WINDOWS" || upper == "META" || upper == "SUPER")
                {
                    modifiers |= NativeMethods.MOD_WIN;
                    continue;
                }

                if (keyName != null)
                {
                    throw new HotkeyException("Hotkey must contain exactly one key");
                }
                keyName = part;
            }

            if (keyName == null)
            {
                throw new HotkeyException("Hotkey must include a non-modifier key");
            }

            KeyToken token = ResolveKey(keyName);
            if (modifiers == 0 && !IsFunctionKey(token.VirtualKey))
            {
                throw new HotkeyException("Use at least one modifier (Ctrl/Alt/Shift/Win) or an F-key");
            }

            return new HotkeySpec(modifiers, token.VirtualKey, Format(modifiers, token.Display));
        }

        public static string Normalize(string text)
        {
            return Parse(text).Text;
        }

        public static string Format(uint modifiers, string keyDisplay)
        {
            List<string> parts = new List<string>();
            if ((modifiers & NativeMethods.MOD_CONTROL) != 0) parts.Add("Ctrl");
            if ((modifiers & NativeMethods.MOD_ALT) != 0) parts.Add("Alt");
            if ((modifiers & NativeMethods.MOD_SHIFT) != 0) parts.Add("Shift");
            if ((modifiers & NativeMethods.MOD_WIN) != 0) parts.Add("Win");
            parts.Add(keyDisplay);
            return string.Join("+", parts.ToArray());
        }

        public static bool IsFunctionKey(uint virtualKey)
        {
            return virtualKey >= 0x70 && virtualKey <= 0x87;
        }

        public static string DisplayFromKey(Keys keyCode)
        {
            if (keyCode >= Keys.A && keyCode <= Keys.Z)
            {
                return ((char)('A' + (int)keyCode - (int)Keys.A)).ToString();
            }
            if (keyCode >= Keys.D0 && keyCode <= Keys.D9)
            {
                return ((char)('0' + (int)keyCode - (int)Keys.D0)).ToString();
            }
            if (keyCode >= Keys.NumPad0 && keyCode <= Keys.NumPad9)
            {
                return "Num" + ((int)keyCode - (int)Keys.NumPad0).ToString(CultureInfo.InvariantCulture);
            }
            if (keyCode >= Keys.F1 && keyCode <= Keys.F24)
            {
                return "F" + ((int)keyCode - (int)Keys.F1 + 1).ToString(CultureInfo.InvariantCulture);
            }

            switch (keyCode)
            {
                case Keys.Back: return "Backspace";
                case Keys.Tab: return "Tab";
                case Keys.Return: return "Enter";
                case Keys.Escape: return "Esc";
                case Keys.Space: return "Space";
                case Keys.PageUp: return "PgUp";
                case Keys.PageDown: return "PgDown";
                case Keys.End: return "End";
                case Keys.Home: return "Home";
                case Keys.Left: return "Left";
                case Keys.Up: return "Up";
                case Keys.Right: return "Right";
                case Keys.Down: return "Down";
                case Keys.Insert: return "Ins";
                case Keys.Delete: return "Del";
                case Keys.PrintScreen: return "Print";
                case Keys.Pause: return "Pause";
                case Keys.Apps: return "Menu";
                case Keys.Help: return "Help";
                case Keys.CapsLock: return "CapsLock";
                case Keys.NumLock: return "NumLock";
                case Keys.Scroll: return "ScrollLock";
                case Keys.OemMinus: return "-";
                case Keys.Oemplus: return "=";
                case Keys.OemOpenBrackets: return "[";
                case Keys.OemCloseBrackets: return "]";
                case Keys.OemPipe: return "\\";
                case Keys.OemSemicolon: return ";";
                case Keys.OemQuotes: return "'";
                case Keys.Oemcomma: return ",";
                case Keys.OemPeriod: return ".";
                case Keys.OemQuestion: return "/";
                case Keys.Oemtilde: return "`";
                case Keys.Add: return "NumAdd";
                case Keys.Subtract: return "NumSubtract";
                case Keys.Multiply: return "NumMultiply";
                case Keys.Divide: return "NumDivide";
                case Keys.Decimal: return "NumDecimal";
                default: return null;
            }
        }

        private static KeyToken ResolveKey(string name)
        {
            string raw = name.Trim();
            KeyToken token;
            if (Aliases.TryGetValue(raw, out token)) return token;

            if (raw.Length == 1)
            {
                char c = char.ToUpperInvariant(raw[0]);
                if (c >= 'A' && c <= 'Z') return new KeyToken((uint)c, c.ToString());
                if (c >= '0' && c <= '9') return new KeyToken((uint)c, c.ToString());
            }

            string upper = raw.ToUpperInvariant();
            if (upper.StartsWith("F", StringComparison.Ordinal) && upper.Length > 1)
            {
                int number;
                if (int.TryParse(upper.Substring(1), NumberStyles.None, CultureInfo.InvariantCulture, out number))
                {
                    if (number >= 1 && number <= 24)
                    {
                        return new KeyToken((uint)(0x70 + number - 1), "F" + number.ToString(CultureInfo.InvariantCulture));
                    }
                }
            }

            throw new HotkeyException("Unsupported key: " + name);
        }
    }

    internal sealed class HotkeyManager : IDisposable
    {
        private readonly HotkeyHost _host;
        private readonly Dictionary<int, string> _registered;
        private int _nextId;

        public event Action<string> HotkeyPressed;

        public HotkeyManager()
        {
            _registered = new Dictionary<int, string>();
            _nextId = 1;
            _host = new HotkeyHost(this);
        }

        public Dictionary<string, string> Apply(List<GroupData> groups, bool paused)
        {
            UnregisterAll();
            Dictionary<string, string> results = new Dictionary<string, string>();
            if (paused)
            {
                foreach (GroupData group in groups) results[group.Id] = "Paused";
                return results;
            }

            Dictionary<string, string> used = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
            foreach (GroupData group in groups)
            {
                if (string.IsNullOrEmpty(group.Hotkey) || group.Hotkey.Trim().Length == 0)
                {
                    results[group.Id] = "No hotkey set";
                    continue;
                }

                HotkeySpec spec;
                try
                {
                    spec = HotkeyParser.Parse(group.Hotkey);
                }
                catch (HotkeyException ex)
                {
                    results[group.Id] = ex.Message;
                    continue;
                }

                if (used.ContainsKey(spec.Text))
                {
                    results[group.Id] = "Conflicts with group '" + used[spec.Text] + "'";
                    continue;
                }
                used[spec.Text] = group.Name;

                int id = _nextId;
                _nextId++;
                bool ok = NativeMethods.RegisterHotKey(
                    _host.Handle,
                    id,
                    spec.Modifiers | NativeMethods.MOD_NOREPEAT,
                    spec.VirtualKey);
                if (!ok)
                {
                    int error = Marshal.GetLastWin32Error();
                    if (error == NativeMethods.ERROR_HOTKEY_ALREADY_REGISTERED)
                    {
                        results[group.Id] = "Hotkey is already in use by another app";
                    }
                    else
                    {
                        results[group.Id] = "Could not register hotkey (error " +
                            error.ToString(CultureInfo.InvariantCulture) + ")";
                    }
                    continue;
                }

                _registered[id] = group.Id;
                results[group.Id] = null;
            }
            return results;
        }

        public void UnregisterAll()
        {
            foreach (int id in new List<int>(_registered.Keys))
            {
                NativeMethods.UnregisterHotKey(_host.Handle, id);
            }
            _registered.Clear();
            _nextId = 1;
        }

        internal void Dispatch(int id)
        {
            string groupId;
            if (_registered.TryGetValue(id, out groupId))
            {
                Action<string> handler = HotkeyPressed;
                if (handler != null) handler(groupId);
            }
        }

        public void Dispose()
        {
            UnregisterAll();
            _host.Destroy();
        }
    }

    internal sealed class HotkeyHost : NativeWindow
    {
        private readonly HotkeyManager _manager;

        public HotkeyHost(HotkeyManager manager)
        {
            _manager = manager;
            CreateParams parameters = new CreateParams();
            parameters.Caption = "PasteNotesHotkeyHost";
            parameters.Parent = new IntPtr(-3);
            CreateHandle(parameters);
        }

        protected override void WndProc(ref Message message)
        {
            if (message.Msg == NativeMethods.WM_HOTKEY)
            {
                _manager.Dispatch(message.WParam.ToInt32());
                return;
            }
            base.WndProc(ref message);
        }

        public void Destroy()
        {
            if (Handle != IntPtr.Zero) DestroyHandle();
        }
    }

    internal sealed class ClipboardSnapshot
    {
        public DataObject Data;
        public bool HasRestorableData;
        public bool IsEmpty;
        public uint SequenceAtCapture;
    }

    internal static class ClipboardService
    {
        public static ClipboardSnapshot Capture()
        {
            try
            {
                IDataObject source = Clipboard.GetDataObject();
                ClipboardSnapshot snapshot = new ClipboardSnapshot();
                snapshot.Data = new DataObject();
                snapshot.SequenceAtCapture = NativeMethods.GetClipboardSequenceNumber();
                snapshot.IsEmpty = true;
                if (source == null) return snapshot;

                string[] formats = source.GetFormats(false);
                if (formats == null || formats.Length == 0) return snapshot;
                snapshot.IsEmpty = false;

                foreach (string format in formats)
                {
                    if (string.IsNullOrEmpty(format) || IsUnsafeFormat(format)) continue;
                    object value = null;
                    try { value = source.GetData(format, false); }
                    catch { continue; }

                    object copy = CopyValue(value);
                    if (copy == null) continue;
                    try
                    {
                        snapshot.Data.SetData(format, copy);
                        snapshot.HasRestorableData = true;
                    }
                    catch
                    {
                    }
                }
                return snapshot;
            }
            catch (Exception ex)
            {
                Log.Warn("Could not snapshot the clipboard: " + ex.Message);
                return null;
            }
        }

        public static bool SetText(string text)
        {
            string value = text ?? "";
            for (int attempt = 0; attempt < 6; attempt++)
            {
                try
                {
                    Clipboard.SetText(value, TextDataFormat.UnicodeText);
                    return Clipboard.ContainsText(TextDataFormat.UnicodeText);
                }
                catch (ExternalException)
                {
                    Thread.Sleep(25);
                }
                catch (Exception ex)
                {
                    Log.Warn("Could not write text to the clipboard: " + ex.Message);
                    return false;
                }
            }
            return false;
        }

        public static bool Clear()
        {
            for (int attempt = 0; attempt < 6; attempt++)
            {
                try
                {
                    Clipboard.Clear();
                    return true;
                }
                catch (ExternalException)
                {
                    Thread.Sleep(25);
                }
                catch (Exception ex)
                {
                    Log.Warn("Could not clear the clipboard: " + ex.Message);
                    return false;
                }
            }
            return false;
        }

        public static bool Restore(ClipboardSnapshot snapshot)
        {
            if (snapshot == null || snapshot.Data == null) return false;
            if (!snapshot.HasRestorableData) return Clear();
            for (int attempt = 0; attempt < 6; attempt++)
            {
                try
                {
                    Clipboard.SetDataObject(snapshot.Data, true);
                    return true;
                }
                catch (ExternalException)
                {
                    Thread.Sleep(25);
                }
                catch (Exception ex)
                {
                    Log.Warn("Could not restore the clipboard: " + ex.Message);
                    return false;
                }
            }
            return false;
        }

        private static bool IsUnsafeFormat(string format)
        {
            return string.Equals(format, DataFormats.MetafilePict, StringComparison.OrdinalIgnoreCase)
                || string.Equals(format, DataFormats.Palette, StringComparison.OrdinalIgnoreCase)
                || string.Equals(format, DataFormats.EnhancedMetafile, StringComparison.OrdinalIgnoreCase)
                || string.Equals(format, DataFormats.Serializable, StringComparison.OrdinalIgnoreCase);
        }

        private static object CopyValue(object value)
        {
            if (value == null) return null;

            string text = value as string;
            if (text != null) return text;

            string[] files = value as string[];
            if (files != null) return (string[])files.Clone();

            byte[] bytes = value as byte[];
            if (bytes != null) return (byte[])bytes.Clone();

            Bitmap bitmap = value as Bitmap;
            if (bitmap != null) return new Bitmap(bitmap);

            Image image = value as Image;
            if (image != null) return new Bitmap(image);

            MemoryStream memory = value as MemoryStream;
            if (memory != null) return memory.ToArray();

            Stream stream = value as Stream;
            if (stream != null)
            {
                using (MemoryStream copy = new MemoryStream())
                {
                    stream.Position = 0;
                    stream.CopyTo(copy);
                    return copy.ToArray();
                }
            }

            return null;
        }
    }

    internal sealed class PasteService : IDisposable
    {
        private ClipboardSnapshot _pending;
        private uint _sequenceAfterSet;
        private System.Windows.Forms.Timer _restoreTimer;

        public string LastError { get; private set; }

        public bool Paste(string text, IntPtr target, bool restoreClipboard)
        {
            LastError = null;

            if (string.IsNullOrEmpty(text))
            {
                LastError = "This template is empty.";
                return false;
            }
            if (target == IntPtr.Zero || !NativeMethods.IsWindow(target))
            {
                LastError = "The target window is no longer open.";
                return false;
            }
            if (NativeMethods.IsOwnedByCurrentProcess(target))
            {
                LastError = "The target window belongs to PasteNotes.";
                return false;
            }
            if (UiHelpers.IsShellWindow(target))
            {
                LastError = "No editable window was selected. Click the window you want to paste into, then use the hotkey again.";
                return false;
            }

            ClipboardSnapshot snapshot = null;
            if (restoreClipboard)
            {
                snapshot = ClipboardService.Capture();
                if (snapshot == null)
                {
                    LastError = "The clipboard is busy. Paste cancelled to avoid losing its contents.";
                    return false;
                }
                if (!snapshot.HasRestorableData && !snapshot.IsEmpty)
                {
                    LastError = "The clipboard contains a format PasteNotes cannot safely preserve. Disable 'Restore previous clipboard' or copy text first.";
                    return false;
                }
            }

            if (!ClipboardService.SetText(text))
            {
                RestoreNow(snapshot);
                LastError = "Could not write the template to the clipboard.";
                return false;
            }

            uint sequence = NativeMethods.GetClipboardSequenceNumber();

            if (!NativeMethods.ForceForeground(target))
            {
                RestoreNow(snapshot);
                LastError = "Could not restore focus to the target window.";
                return false;
            }

            Thread.Sleep(70);

            if (NativeMethods.GetForegroundWindow() != target)
            {
                RestoreNow(snapshot);
                LastError = "The target window changed before the paste could be sent.";
                return false;
            }

            if (!NativeMethods.SendPaste())
            {
                RestoreNow(snapshot);
                LastError = "Windows did not accept the paste keystroke.";
                return false;
            }

            if (restoreClipboard && snapshot != null)
            {
                ScheduleRestore(snapshot, sequence);
            }
            return true;
        }

        public void RestorePending()
        {
            CancelRestoreTimer();
            if (_pending != null)
            {
                if (NativeMethods.GetClipboardSequenceNumber() == _sequenceAfterSet)
                {
                    ClipboardService.Restore(_pending);
                }
                _pending = null;
            }
        }

        private void ScheduleRestore(ClipboardSnapshot snapshot, uint sequence)
        {
            CancelRestoreTimer();
            _pending = snapshot;
            _sequenceAfterSet = sequence;
            _restoreTimer = new System.Windows.Forms.Timer();
            _restoreTimer.Interval = 350;
            _restoreTimer.Tick += delegate
            {
                CancelRestoreTimer();
                if (_pending == null) return;
                if (NativeMethods.GetClipboardSequenceNumber() == _sequenceAfterSet)
                {
                    ClipboardService.Restore(_pending);
                }
                else
                {
                    Log.Info("Clipboard changed during paste; skipped restoring the previous contents.");
                }
                _pending = null;
            };
            _restoreTimer.Start();
        }

        private void CancelRestoreTimer()
        {
            if (_restoreTimer != null)
            {
                _restoreTimer.Stop();
                _restoreTimer.Dispose();
                _restoreTimer = null;
            }
        }

        private static void RestoreNow(ClipboardSnapshot snapshot)
        {
            if (snapshot != null)
            {
                ClipboardService.Restore(snapshot);
            }
        }

        public void Dispose()
        {
            RestorePending();
        }
    }

        public static TextFormatFlags TextFlags(string text, bool centerVertically)
        {
            TextFormatFlags flags = TextFormatFlags.NoPrefix | TextFormatFlags.EndEllipsis;
            if (centerVertically) flags |= TextFormatFlags.VerticalCenter;
            if (RtlHelper.IsRtl(text))
            {
                flags |= TextFormatFlags.Right | TextFormatFlags.RightToLeft;
            }
            else
            {
                flags |= TextFormatFlags.Left;
            }
            return flags;
        }

        public static string Preview(string content, int limit)
        {
            if (string.IsNullOrEmpty(content)) return "";
            System.Text.StringBuilder builder = new System.Text.StringBuilder();
            bool lastSpace = false;
            for (int index = 0; index < content.Length; index++)
            {
                char c = content[index];
                if (char.IsWhiteSpace(c))
                {
                    if (!lastSpace && builder.Length > 0) builder.Append(' ');
                    lastSpace = true;
                }
                else
                {
                    builder.Append(c);
                    lastSpace = false;
                }
                if (builder.Length > limit) break;
            }
            string text = builder.ToString().Trim();
            if (text.Length > limit) text = text.Substring(0, Math.Max(0, limit - 3)) + "...";
            return text;
        }

            Cursor = Cursors.Hand;

            if (!Enabled)
            {
                using (SolidBrush gray = new SolidBrush(Theme.BorderSoft))
                {
                    e.Graphics.FillEllipse(gray, circle);
                }
                return;
            }

            using (SolidBrush brush = new SolidBrush(Palette.ColorValue(ColorKey)))
            {
                e.Graphics.FillEllipse(brush, circle);
            }

            if (_selected)
            {
                using (Pen pen = new Pen(Color.White, 2f))
                {
                    e.Graphics.DrawEllipse(pen, circle);
                }
            }
            else if (_hover || Focused)
            {
                using (Pen pen = new Pen(Theme.Muted, 1.4f))
                {
                    e.Graphics.DrawEllipse(pen, circle);
                }
            }
        }

        protected override void OnMouseEnter(EventArgs e)
        {
            _hover = true;
            Invalidate();
            base.OnMouseEnter(e);
        }

        protected override void OnMouseLeave(EventArgs e)
        {
            _hover = false;
            Invalidate();
            base.OnMouseLeave(e);
        }

        protected override void OnMouseDown(MouseEventArgs e)
        {
            base.OnMouseDown(e);
            if (e.Button != MouseButtons.Left || !Enabled) return;
            Focus();
            if (Picked != null) Picked(this, EventArgs.Empty);
        }

        protected override bool IsInputKey(Keys keyData)
        {
            if (keyData == Keys.Space || keyData == Keys.Enter) return true;
            return base.IsInputKey(keyData);
        }

        protected override void OnKeyDown(KeyEventArgs e)
        {
            if ((e.KeyCode == Keys.Space || e.KeyCode == Keys.Enter) && Enabled)
            {
                e.Handled = true;
                e.SuppressKeyPress = true;
                if (Picked != null) Picked(this, EventArgs.Empty);
                return;
            }
            base.OnKeyDown(e);
        }
    }

            BackColor = Theme.InputBack;
            ForeColor = Theme.Text;
            BorderStyle = BorderStyle.None;
        }

        protected override bool ProcessCmdKey(ref Message msg, Keys keyData)
        {
            return base.ProcessCmdKey(ref msg, keyData);
        }

        protected override void OnKeyDown(KeyEventArgs e)
        {
            Keys key = e.KeyCode;
            bool functionKey = key >= Keys.F1 && key <= Keys.F24;

            if (e.Control || e.Alt || functionKey)
            {
                string display = HotkeyParser.DisplayFromKey(key);
                if (display != null)
                {
                    uint modifiers = 0;
                    if (e.Control) modifiers |= NativeMethods.MOD_CONTROL;
                    if (e.Alt) modifiers |= NativeMethods.MOD_ALT;
                    if (e.Shift) modifiers |= NativeMethods.MOD_SHIFT;

                    string candidate = HotkeyParser.Format(modifiers, display);
                    try
                    {
                        HotkeySpec spec = HotkeyParser.Parse(candidate);
                        Text = spec.Text;
                        SelectionStart = Text.Length;
                        SelectionLength = 0;
                        if (CaptureCommitted != null) CaptureCommitted(this, EventArgs.Empty);
                    }
                    catch (HotkeyException)
                    {
                    }

                    e.Handled = true;
                    e.SuppressKeyPress = true;
                    return;
                }
            }

            base.OnKeyDown(e);
        }
    }

        private void ReloadGroups(string selectId)
        {
            _loading = true;
            _groupList.BeginUpdate();
            _groupList.Items.Clear();
            foreach (GroupData group in _store.Data.Groups)
            {
                _groupList.Items.Add(group);
            }
            _groupList.EndUpdate();
            _loading = false;

            if (_store.Data.Groups.Count == 0)
            {
                SelectGroup(null);
                return;
            }

            int row = 0;
            if (!string.IsNullOrEmpty(selectId))
            {
                for (int index = 0; index < _groupList.Items.Count; index++)
                {
                    GroupData item = _groupList.Items[index] as GroupData;
                    if (item != null && item.Id == selectId)
                    {
                        row = index;
                        break;
                    }
                }
            }

            _loading = true;
            _groupList.SelectedIndex = row;
            _loading = false;
            SelectGroup(_store.Data.Groups[row]);
        }

                        row = index;

            _loading = false;

        private void HotkeyKeyDown(object sender, KeyEventArgs e)
        {
            if (e.KeyCode == Keys.Enter)
            {
                e.Handled = true;
                e.SuppressKeyPress = true;
                CommitHotkey(true);
            }
            else if (e.KeyCode == Keys.Escape)
            {
                e.Handled = true;
                e.SuppressKeyPress = true;
                if (_group != null)
                {
                    _loading = true;
                    _hotkeyBox.Text = _group.Hotkey;
                    _loading = false;
                }
                RefreshHotkeyStatus();
            }
        }

        private void CommitHotkey(bool showMessage)
        {
            if (_loading || _group == null) return;

            string raw = _hotkeyBox.Text.Trim();
            string normalized;
            if (raw.Length == 0)
            {
                normalized = "";
            }
            else
            {
                try
                {
                    normalized = HotkeyParser.Normalize(raw);
                }
                catch (HotkeyException ex)
                {
                    _loading = true;
                    _hotkeyBox.Text = _group.Hotkey;
                    _loading = false;
                    _hotkeyStatus.Text = "Invalid hotkey";
                    _hotkeyStatus.ForeColor = Theme.Danger;
                    if (showMessage)
                    {
                        SetStatus("Hotkey not changed: " + ex.Message);
                    }
                    return;
                }
            }

            _loading = true;
            _hotkeyBox.Text = normalized;
            _loading = false;

            if (normalized == _group.Hotkey)
            {
                RefreshHotkeyStatus();
                return;
            }

            _group.Hotkey = normalized;
            _hotkeyStatus.Text = normalized.Length == 0 ? "Not set" : "Checking...";
            _hotkeyStatus.ForeColor = Theme.Muted;
            ScheduleSave();
            _hotkeyTimer.Stop();
            _hotkeyTimer.Start();
            if (showMessage)
            {
                SetStatus(normalized.Length == 0 ? "Hotkey cleared." : "Hotkey set to " + normalized + ".");
            }
        }

        private void SwatchPicked(object sender, EventArgs e)
        {
            if (_loading || _group == null) return;
            ColorSwatch swatch = sender as ColorSwatch;
            if (swatch == null) return;
            _group.Color = Palette.Normalize(swatch.ColorKey);
            RefreshColorSwatches();
            _groupList.Invalidate();
            ScheduleSave();
            SetStatus("Color set to " + Palette.Label(_group.Color) + ".");
        }

        private void RefreshColorSwatches()
        {
            string selected = _group == null ? "" : Palette.Normalize(_group.Color);
            foreach (KeyValuePair<string, ColorSwatch> pair in _swatches)
            {
                pair.Value.Enabled = _group != null;
                pair.Value.IsSelected = _group != null && string.Equals(pair.Key, selected, StringComparison.OrdinalIgnoreCase);
                pair.Value.Invalidate();
            }
            _colorLabel.Enabled = _group != null;
        }

        public void SetHotkeyStatuses(Dictionary<string, string> statuses)
        {
            _statuses = statuses ?? new Dictionary<string, string>();
            RefreshHotkeyStatus();
        }

        private void RefreshHotkeyStatus()
        {
            if (_hotkeyStatus == null) return;
            if (_group == null)
            {
                _hotkeyStatus.Text = "";
                _hotkeyStatus.ForeColor = Theme.Muted;
                _hotkeyStatus.Tag = null;
                return;
            }

            string message;
            if (!_statuses.TryGetValue(_group.Id, out message))
            {
                _hotkeyStatus.Text = "";
                _hotkeyStatus.ForeColor = Theme.Muted;
                _hotkeyStatus.Tag = null;
                return;
            }

            if (message == null)
            {
                _hotkeyStatus.Text = "Active";
                _hotkeyStatus.ForeColor = Theme.Success;
                _hotkeyStatus.Tag = null;
            }
            else if (message == "No hotkey set")
            {
                _hotkeyStatus.Text = "Not set";
                _hotkeyStatus.ForeColor = Theme.Warning;
                _hotkeyStatus.Tag = null;
            }
            else if (message == "Paused")
            {
                _hotkeyStatus.Text = "Paused";
                _hotkeyStatus.ForeColor = Theme.Muted;
                _hotkeyStatus.Tag = null;
            }
            else
            {
                _hotkeyStatus.Text = message;
                _hotkeyStatus.ForeColor = Theme.Danger;
                _hotkeyStatus.Tag = message;
            }

            if (_hotkeyStatus.Tag != null)
            {
                _savedTip.SetToolTip(_hotkeyStatus, _hotkeyStatus.Tag.ToString());
            }
            else
            {
                _savedTip.SetToolTip(_hotkeyStatus, "");
            }
        }

        private void GroupListDrawItem(object sender, DrawItemEventArgs e)
        {
            if (e.Index < 0) return;
            GroupData group = _groupList.Items[e.Index] as GroupData;
            if (group == null) return;

            bool selected = (e.State & DrawItemState.Selected) != 0;
            Color back = selected ? Theme.RowSelected : Theme.InputBack;
            using (SolidBrush brush = new SolidBrush(back))
            {
                e.Graphics.FillRectangle(brush, e.Bounds);
            }

            int dotSize = 12;
            Rectangle dot = new Rectangle(
                e.Bounds.X + 9,
                e.Bounds.Y + Math.Max(0, (e.Bounds.Height - dotSize) / 2),
                dotSize,
                dotSize);
            e.Graphics.SmoothingMode = System.Drawing.Drawing2D.SmoothingMode.AntiAlias;
            using (SolidBrush brush = new SolidBrush(Palette.ColorValue(group.Color)))
            {
                e.Graphics.FillEllipse(brush, dot);
            }

            Rectangle textBounds = new Rectangle(
                e.Bounds.X + 28,
                e.Bounds.Y,
                Math.Max(1, e.Bounds.Width - 34),
                e.Bounds.Height);
            TextFormatFlags flags = UiDraw.TextFlags(group.Name, true);
            TextRenderer.DrawText(
                e.Graphics,
                group.Name,
                _groupList.Font,
                textBounds,
                selected ? Color.White : Theme.Text,
                flags);
        }

        private void TemplateListDrawItem(object sender, DrawItemEventArgs e)
        {
            if (e.Index < 0) return;
            TemplateData template = _templateList.Items[e.Index] as TemplateData;
            if (template == null) return;

            bool selected = (e.State & DrawItemState.Selected) != 0;
            Color back = selected ? Theme.RowSelected : Theme.InputBack;
            using (SolidBrush brush = new SolidBrush(back))
            {
                e.Graphics.FillRectangle(brush, e.Bounds);
            }

            Color accent = _group == null ? Theme.Muted : Palette.ColorValue(_group.Color);
            using (SolidBrush brush = new SolidBrush(accent))
            {
                int barHeight = Math.Max(8, e.Bounds.Height - 14);
                e.Graphics.FillRectangle(
                    brush,
                    new Rectangle(e.Bounds.X + 9, e.Bounds.Y + (e.Bounds.Height - barHeight) / 2, 4, barHeight));
            }

            string title = string.IsNullOrEmpty(template.Title) ? "Untitled template" : template.Title;
            Rectangle textBounds = new Rectangle(
                e.Bounds.X + 20,
                e.Bounds.Y,
                Math.Max(1, e.Bounds.Width - 26),
                e.Bounds.Height);
            TextRenderer.DrawText(
                e.Graphics,
                title,
                _templateList.Font,
                textBounds,
                selected ? Color.White : Theme.Text,
                UiDraw.TextFlags(title, true));
        }

        private void AddGroup()
        {
            bool accepted;
            string initial = _store.UniqueGroupName("New group");
            string value = TextPromptForm.Prompt(this, "New group", "Group name:", initial, out accepted);
            if (!accepted) return;

            string name = value.Trim();
            if (name.Length == 0) name = _store.UniqueGroupName("New group");
            GroupData group = _store.AddGroup(name);
            ReloadGroups(group.Id);
            ScheduleSave();
            if (HotkeysChanged != null) HotkeysChanged(this, EventArgs.Empty);
            SetStatus("Added group '" + group.Name + "'. Set a hotkey for it.");
        }

        private void RenameGroup()
        {
            if (_group == null) return;
            bool accepted;
            string value = TextPromptForm.Prompt(this, "Rename group", "Group name:", _group.Name, out accepted);
            if (!accepted) return;

            string name = value.Trim();
            if (name.Length == 0) return;
            _group.Name = name;
            ReloadGroups(_group.Id);
            ScheduleSave();
            SetStatus("Group renamed.");
        }

        private void DeleteGroup()
        {
            if (_group == null) return;
            DialogResult answer = MessageBox.Show(
                this,
                "Delete group '" + _group.Name + "' and all of its templates?",
                "Delete group",
                MessageBoxButtons.YesNo,
                MessageBoxIcon.Warning,
                MessageBoxDefaultButton.Button2);
            if (answer != DialogResult.Yes) return;

            string name = _group.Name;
            _store.RemoveGroup(_group.Id);
            _group = null;
            _template = null;
            ReloadGroups(null);
            ScheduleSave();
            if (HotkeysChanged != null) HotkeysChanged(this, EventArgs.Empty);
            SetStatus("Deleted group '" + name + "'.");
        }

        private void MoveGroup(int delta)
        {
            if (_group == null) return;
            if (_store.MoveGroup(_group.Id, delta))
            {
                ReloadGroups(_group.Id);
                ScheduleSave();
            }
        }

        private void AddTemplate()
        {
            if (_group == null) return;
            TemplateData template = _store.AddTemplate(_group);
            if (template == null) return;
            ReloadTemplates(template.Id);
            ScheduleSave();
            _titleBox.Focus();
            _titleBox.SelectAll();
            SetStatus("Added a template. Type a title and its text.");
        }

        private void DuplicateTemplate()
        {
            if (_group == null || _template == null) return;
            string sourceTitle = _template.Title;
            TemplateData copy = _store.DuplicateTemplate(_group, _template);
            if (copy == null) return;
            ReloadTemplates(copy.Id);
            ScheduleSave();
            SetStatus("Duplicated template '" + sourceTitle + "'.");
        }

        private void DeleteTemplate()
        {
            if (_group == null || _template == null) return;
            DialogResult answer = MessageBox.Show(
                this,
                "Delete template '" + _template.Title + "'?",
                "Delete template",
                MessageBoxButtons.YesNo,
                MessageBoxIcon.Warning,
                MessageBoxDefaultButton.Button2);
            if (answer != DialogResult.Yes) return;

            string title = _template.Title;
            _store.RemoveTemplate(_group, _template.Id);
            _template = null;
            ReloadTemplates(null);
            ScheduleSave();
            SetStatus("Deleted template '" + title + "'.");
        }

        private void MoveTemplate(int delta)
        {
            if (_group == null || _template == null) return;
            if (_store.MoveTemplate(_group, _template, delta))
            {
                ReloadTemplates(_template.Id);
                ScheduleSave();
            }
        }

        private void ScheduleSave()
        {
            SetSavedState("Saving...", Theme.Warning);
            _saveTimer.Stop();
            _saveTimer.Start();
        }

        private void SaveNow()
        {
            _saveTimer.Stop();
            try
            {
                _store.Save();
                SetSavedState("All changes saved", Theme.Muted);
                _savedTip.SetToolTip(_savedLabel, "Changes are saved automatically.");
            }
            catch (Exception ex)
            {
                Log.Error("Could not save PasteNotes data", ex);
                SetSavedState("Save failed - click to retry", Theme.Danger);
                _savedTip.SetToolTip(_savedLabel, ex.Message);
                SetStatus("Could not save data: " + ex.Message);
            }
        }

        private void SetSavedState(string text, Color color)
        {
            _savedLabel.Text = text;
            _savedLabel.ForeColor = color;
        }

        private void SetStatus(string message)
        {
            _statusLabel.Text = message;
            _statusTimer.Stop();
            _statusTimer.Start();
        }

        public void FlushSave()
        {
            _saveTimer.Stop();
            SaveNow();
        }

        public void PrepareExit()
        {
            _allowClose = true;
            _saveTimer.Stop();
            _hotkeyTimer.Stop();
            _statusTimer.Stop();
            SaveNow();
        }

        public void ShowManager()
        {
            if (!Visible) Show();
            if (WindowState == FormWindowState.Minimized) WindowState = FormWindowState.Normal;
            BringToFront();
            Activate();
            NativeMethods.ForceForeground(Handle);
        }

        protected override void OnFormClosing(FormClosingEventArgs e)
        {
            _saveTimer.Stop();
            SaveNow();
            if (!_allowClose)
            {
                e.Cancel = true;
                Hide();
                if (HiddenToTray != null) HiddenToTray(this, EventArgs.Empty);
                return;
            }
            base.OnFormClosing(e);
        }

        protected override void OnShown(EventArgs e)
        {
            base.OnShown(e);
            if (_groupList != null && _groupList.SelectedIndex >= 0)
            {
                _groupList.Focus();
            }
        }

        protected override void Dispose(bool disposing)
        {
            if (disposing)
            {
                if (_saveTimer != null) _saveTimer.Dispose();
                if (_hotkeyTimer != null) _hotkeyTimer.Dispose();
                if (_statusTimer != null) _statusTimer.Dispose();
                if (_savedTip != null) _savedTip.Dispose();
            }
            base.Dispose(disposing);
        }
    }

        public PickerForm(GroupData group)

        public void OpenAtCursor()
        {
            Point cursor = Cursor.Position;
            Screen screen = Screen.FromPoint(cursor);
            Rectangle area = screen.WorkingArea;
            int width = Math.Min(Width, area.Width);
            int height = Math.Min(Height, area.Height);

            int x = cursor.X + 14;
            int y = cursor.Y + 18;
            if (x + width > area.Right) x = area.Right - width;
            if (y + height > area.Bottom) y = cursor.Y - height - 16;
            x = Math.Max(area.Left, Math.Min(x, area.Right - width));
            y = Math.Max(area.Top, Math.Min(y, area.Bottom - height));

            Location = new Point(x, y);
            _shownAt = Environment.TickCount;
            Show();
            NativeMethods.ForceForeground(Handle);
            _search.Focus();
            _search.SelectAll();
        }

        public void DismissSilently()
        {
            _finished = true;
            Hide();
        }

        private void ApplyFilter()
        {
            string query = _search.Text.Trim();
            _matches.Clear();
            foreach (TemplateData template in _group.Templates)
            {
                if (query.Length == 0)
                {
                    _matches.Add(template);
                    continue;
                }

                string title = template.Title ?? "";
                string content = template.Content ?? "";
                if (title.IndexOf(query, StringComparison.OrdinalIgnoreCase) >= 0 ||
                    content.IndexOf(query, StringComparison.OrdinalIgnoreCase) >= 0)
                {
                    _matches.Add(template);
                }
            }

            _list.BeginUpdate();
            _list.Items.Clear();
            if (_matches.Count == 0)
            {
                _emptyMessage = _group.Templates.Count == 0
                    ? "No templates in this group"
                    : "No matching templates";
                _emptyLabel.Text = _emptyMessage;
                _emptyLabel.Visible = true;
                _emptyLabel.BringToFront();
            }
            else
            {
                _emptyLabel.Visible = false;
                foreach (TemplateData template in _matches)
                {
                    _list.Items.Add(new PickerRow(
                        template,
                        UiDraw.Preview(template.Content, 140)));
                }
            }
            _list.EndUpdate();

            if (_matches.Count > 0)
            {
                _list.SelectedIndex = 0;
                _list.TopIndex = 0;
            }
            else
            {
                _list.ClearSelected();
            }
        }

        private void SearchKeyDown(object sender, KeyEventArgs e)
        {
            if (e.KeyCode == Keys.Down)
            {
                MoveCurrent(1);
                e.Handled = true;
                e.SuppressKeyPress = true;
            }
            else if (e.KeyCode == Keys.Up)
            {
                MoveCurrent(-1);
                e.Handled = true;
                e.SuppressKeyPress = true;
            }
            else if (e.KeyCode == Keys.PageDown)
            {
                MoveCurrent(8);
                e.Handled = true;
                e.SuppressKeyPress = true;
            }
            else if (e.KeyCode == Keys.PageUp)
            {
                MoveCurrent(-8);
                e.Handled = true;
                e.SuppressKeyPress = true;
            }
            else if (e.KeyCode == Keys.Enter)
            {
                AcceptCurrent();
                e.Handled = true;
                e.SuppressKeyPress = true;
            }
            else if (e.KeyCode == Keys.Escape)
            {
                Cancel(true);
                e.Handled = true;
                e.SuppressKeyPress = true;
            }
        }

        private void MoveCurrent(int delta)
        {
            if (_matches.Count == 0) return;
            int current = _list.SelectedIndex;
            if (current < 0 || current >= _matches.Count) current = 0;
            int target = Math.Max(0, Math.Min(_matches.Count - 1, current + delta));
            _list.SelectedIndex = target;

            int visible = Math.Max(1, _list.ClientSize.Height / Math.Max(1, _list.ItemHeight));
            int top = _list.TopIndex;
            if (target < top)
            {
                _list.TopIndex = target;
            }
            else if (target >= top + visible)
            {
                _list.TopIndex = Math.Max(0, target - visible + 1);
            }
        }

        private void AcceptCurrent()
        {
            if (_list.SelectedIndex < 0 || _list.SelectedIndex >= _list.Items.Count) return;
            PickerRow row = _list.Items[_list.SelectedIndex] as PickerRow;
            if (row == null || row.Template == null) return;
            Accept(row.Template);
        }

        private void Accept(TemplateData template)
        {
            if (_finished) return;
            _finished = true;
            Hide();
            Action<string> handler = TemplateSelected;
            if (handler != null) handler(template.Content);
        }

        private void Cancel(bool refocus)
        {
            if (_finished) return;
            _finished = true;
            Hide();
            Action<bool> handler = Cancelled;
            if (handler != null) handler(refocus);
        }

        private void ListMouseUp(object sender, MouseEventArgs e)
        {
            if (e.Button != MouseButtons.Left || _finished) return;
            int index = _list.IndexFromPoint(e.Location);
            if (index < 0 || index >= _list.Items.Count)
            {
                _search.Focus();
                return;
            }

            _list.SelectedIndex = index;
            PickerRow row = _list.Items[index] as PickerRow;
            if (row == null || row.Template == null)
            {
                _search.Focus();
                return;
            }
            Accept(row.Template);
        }

        private void ListDrawItem(object sender, DrawItemEventArgs e)
        {
            if (e.Index < 0 || e.Index >= _list.Items.Count) return;
            PickerRow row = _list.Items[e.Index] as PickerRow;
            if (row == null || row.Template == null) return;

            bool selected = (e.State & DrawItemState.Selected) != 0 && row.Template != null;
            using (SolidBrush background = new SolidBrush(Theme.WindowBack))
            {
                e.Graphics.FillRectangle(background, e.Bounds);
            }

            Rectangle rowRect = new Rectangle(
                e.Bounds.X + 3,
                e.Bounds.Y + 2,
                Math.Max(1, e.Bounds.Width - 6),
                Math.Max(1, e.Bounds.Height - 4));
            if (selected)
            {
                UiDraw.FillRounded(e.Graphics, rowRect, 6, _accent);
            }

            string title = string.IsNullOrEmpty(row.Template.Title)
                ? "Untitled template"
                : row.Template.Title;
            bool titleRtl = RtlHelper.IsRtl(title);
            TextFormatFlags titleFlags = TextFormatFlags.NoPrefix |
                                        TextFormatFlags.EndEllipsis |
                                        TextFormatFlags.SingleLine |
                                        TextFormatFlags.VerticalCenter;
            if (titleRtl)
            {
                titleFlags |= TextFormatFlags.Right | TextFormatFlags.RightToLeft;
            }
            else
            {
                titleFlags |= TextFormatFlags.Left;
            }

            Rectangle titleBounds = new Rectangle(
                e.Bounds.X + 14,
                e.Bounds.Y + 4,
                Math.Max(1, e.Bounds.Width - 28),
                22);
            TextRenderer.DrawText(
                e.Graphics,
                title,
                Theme.PickerTitleFont,
                titleBounds,
                selected ? _onAccent : Theme.Text,
                titleFlags);

            if (row.Preview.Length > 0)
            {
                bool previewRtl = RtlHelper.IsRtl(row.Preview);
                TextFormatFlags previewFlags = TextFormatFlags.NoPrefix |
                                               TextFormatFlags.EndEllipsis |
                                               TextFormatFlags.SingleLine |
                                               TextFormatFlags.VerticalCenter;
                if (previewRtl)
                {
                    previewFlags |= TextFormatFlags.Right | TextFormatFlags.RightToLeft;
                }
                else
                {
                    previewFlags |= TextFormatFlags.Left;
                }

                Rectangle previewBounds = new Rectangle(
                    e.Bounds.X + 14,
                    e.Bounds.Y + 25,
                    Math.Max(1, e.Bounds.Width - 28),
                    Math.Max(12, e.Bounds.Height - 28));
                TextRenderer.DrawText(
                    e.Graphics,
                    row.Preview,
                    Theme.SmallFont,
                    previewBounds,
                    selected ? Color.FromArgb(165, _onAccent) : Theme.Muted,
                    previewFlags);
            }
        }

        protected override void OnDeactivate(EventArgs e)
        {
            base.OnDeactivate(e);
            if (_finished) return;
            if (Environment.TickCount - _shownAt < 200) return;
            Cancel(false);
        }

        protected override void OnFormClosing(FormClosingEventArgs e)
        {
            if (!_finished)
            {
                _finished = true;
                Action<bool> handler = Cancelled;
                if (handler != null) handler(true);
            }
            base.OnFormClosing(e);
        }

        protected override void OnResize(EventArgs e)
        {
            base.OnResize(e);
            if (WindowState == FormWindowState.Minimized) return;
            if (Width <= 0 || Height <= 0) return;
            using (GraphicsPath path = UiHelpers.RoundRect(new Rectangle(0, 0, Width, Height), 10))
            {
                Region previous = Region;
                Region = new Region(path);
                if (previous != null) previous.Dispose();
            }
        }

        protected override void OnPaint(PaintEventArgs e)
        {
            base.OnPaint(e);
            e.Graphics.SmoothingMode = SmoothingMode.AntiAlias;
            using (GraphicsPath path = UiHelpers.RoundRect(new Rectangle(0, 0, Width - 1, Height - 1), 10))
            using (Pen pen = new Pen(_accent, 1.4f))
            {
                e.Graphics.DrawPath(pen, path);
            }
        }
    }

        private IntPtr _target;
        private bool _paused;
        private bool _exiting;
        private bool _cleaned;
        private bool _trayHintShown;

        public PasteNotesContext(string[] args)
        {
            bool firstRun = !File.Exists(AppPaths.DataFile);

            _store = new DataStore(AppPaths.DataFile);
            _store.Load();

            _manager = new ManagerForm(_store);
            _manager.HotkeysChanged += delegate { ApplyHotkeys(false); };
            _manager.HiddenToTray += ManagerHiddenToTray;
            IntPtr managerHandle = _manager.Handle;
            if (managerHandle == IntPtr.Zero)
            {
                Log.Warn("The manager window handle was not created during startup.");
            }

            _ui = SynchronizationContext.Current;
            _hotkeys = new HotkeyManager();
            _hotkeys.HotkeyPressed += OnHotkey;
            _paste = new PasteService();

            _activeIcon = IconFactory.Create(false);
            _pausedIcon = IconFactory.Create(true);

            _menu = new ContextMenuStrip();
            _menu.Font = Theme.BaseFont;
            _menu.BackColor = Theme.PanelBack;
            _menu.ForeColor = Theme.Text;
            _menu.ShowImageMargin = false;
            _menu.ShowCheckMargin = true;

            ToolStripMenuItem openItem = new ToolStripMenuItem("Open PasteNotes");
            openItem.Click += delegate { ShowManager(); };
            _pauseItem = new ToolStripMenuItem("Pause hotkeys");
            _pauseItem.Click += delegate { SetPaused(!_paused); };
            ToolStripMenuItem exitItem = new ToolStripMenuItem("Exit");
            exitItem.Click += delegate { ExitApp(); };
            _menu.Items.Add(openItem);
            _menu.Items.Add(new ToolStripSeparator());
            _menu.Items.Add(_pauseItem);
            _menu.Items.Add(new ToolStripSeparator());
            _menu.Items.Add(exitItem);

            _tray = new NotifyIcon();
            _tray.Icon = _activeIcon;
            _tray.Text = "PasteNotes - snippet hotkeys";
            _tray.ContextMenuStrip = _menu;
            _tray.Visible = true;
            _tray.MouseClick += TrayMouseClick;

            CreateShowEvent();
            ApplyHotkeys(true);

            if (HasArgument(args, "--show") || firstRun)
            {
                StartOnUi(delegate { ShowManager(); });
            }
        }

        private static bool HasArgument(string[] args, string name)
        {
            if (args == null) return false;
            foreach (string arg in args)
            {
                if (string.Equals(arg, name, StringComparison.OrdinalIgnoreCase)) return true;
            }
            return false;
        }

        private void CreateShowEvent()
        {
            try
            {
                bool createdNew;
                _showEvent = new EventWaitHandle(
                    false,
                    EventResetMode.AutoReset,
                    SingleInstance.ShowEventName,
                    out createdNew);
                _showWait = ThreadPool.RegisterWaitForSingleObject(
                    _showEvent,
                    ShowRequested,
                    null,
                    Timeout.Infinite,
                    false);
            }
            catch (Exception ex)
            {
                Log.Warn("Could not create the single-instance show event: " + ex.Message);
            }
        }

        private void ShowRequested(object state, bool timedOut)
        {
            if (_exiting) return;
            StartOnUi(delegate { ShowManager(); });
        }

        private void StartOnUi(Action action)
        {
            if (action == null || _exiting) return;
            SynchronizationContext context = _ui;
            if (context != null)
            {
                context.Post(delegate { action(); }, null);
                return;
            }
            try
            {
                _manager.BeginInvoke(action);
            }
            catch
            {
            }
        }

        private void TrayMouseClick(object sender, MouseEventArgs e)
        {
            if (e.Button == MouseButtons.Left) ShowManager();
        }

        private void ManagerHiddenToTray(object sender, EventArgs e)
        {
            if (_trayHintShown) return;
            _trayHintShown = true;
            ShowBalloon(
                "PasteNotes",
                "PasteNotes is still running. Use the tray icon to open it again.",
                ToolTipIcon.Info);
        }

        public void ShowManager()
        {
            if (_exiting) return;
            _manager.ShowManager();
        }

        private void ApplyHotkeys(bool notify)
        {
            if (_paused)
            {
                _hotkeys.UnregisterAll();
                _manager.SetHotkeyStatuses(PausedStatuses());
                return;
            }

            Dictionary<string, string> results = _hotkeys.Apply(_store.Data.Groups, false);
            _manager.SetHotkeyStatuses(results);

            if (!notify) return;

            List<string> problems = new List<string>();
            foreach (GroupData group in _store.Data.Groups)
            {
                string message;
                if (!results.TryGetValue(group.Id, out message)) continue;
                if (message == null || message == "No hotkey set") continue;
                problems.Add(group.Name + ": " + message);
            }

            if (problems.Count == 0) return;
            System.Text.StringBuilder builder = new System.Text.StringBuilder();
            int count = Math.Min(4, problems.Count);
            for (int index = 0; index < count; index++)
            {
                if (index > 0) builder.AppendLine();
                builder.Append(problems[index]);
            }
            if (problems.Count > count) builder.AppendLine("...");
            ShowBalloon("PasteNotes hotkey problem", builder.ToString(), ToolTipIcon.Warning);
        }

        private Dictionary<string, string> PausedStatuses()
        {
            Dictionary<string, string> statuses = new Dictionary<string, string>();
            foreach (GroupData group in _store.Data.Groups)
            {
                statuses[group.Id] = "Paused";
            }
            return statuses;
        }

        private void SetPaused(bool paused)
        {
            if (_paused == paused) return;
            _paused = paused;
            _pauseItem.Checked = paused;
            _tray.Icon = paused ? _pausedIcon : _activeIcon;

            if (paused)
            {
                _hotkeys.UnregisterAll();
                _manager.SetHotkeyStatuses(PausedStatuses());
                ShowBalloon("PasteNotes", "Hotkeys paused", ToolTipIcon.Info);
            }
            else
            {
                ApplyHotkeys(true);
                ShowBalloon("PasteNotes", "Hotkeys resumed", ToolTipIcon.Info);
            }
        }

        private void OnHotkey(string groupId)
        {
            if (_paused || _exiting) return;
            GroupData group = _store.FindGroup(groupId);
            if (group == null) return;

            _target = NativeMethods.GetForegroundWindow();
            ClosePickerSilently();

            PickerForm picker = new PickerForm(group);
            picker.TemplateSelected += OnTemplateSelected;
            picker.Cancelled += OnPickerCancelled;
            _picker = picker;
            picker.OpenAtCursor();
        }

        private void ClosePickerSilently()
        {
            PickerForm previous = _picker;
            _picker = null;
            if (previous == null) return;
            try
            {
                previous.DismissSilently();
                previous.Dispose();
            }
            catch
            {
            }
        }

        private void OnTemplateSelected(string text)
        {
            PickerForm picker = _picker;
            _picker = null;
            IntPtr target = _target;
            bool restore = _store.GetBoolSetting("restore_clipboard", true);

            StartOnUi(delegate
            {
                if (picker != null)
                {
                    try { picker.Dispose(); }
                    catch { }
                }
                PasteAndReport(text, target, restore);
            });
        }

        private void PasteAndReport(string text, IntPtr target, bool restore)
        {
            if (_exiting) return;
            bool pasted = _paste.Paste(text, target, restore);
            if (pasted) return;
            string error = _paste.LastError;
            if (string.IsNullOrEmpty(error)) error = "The template could not be pasted.";
            Log.Warn("Paste failed: " + error);
            ShowBalloon("PasteNotes - paste failed", error, ToolTipIcon.Warning);
        }

        private void OnPickerCancelled(bool refocus)
        {
            PickerForm picker = _picker;
            _picker = null;
            IntPtr target = _target;

            StartOnUi(delegate
            {
                if (picker != null)
                {
                    try { picker.Dispose(); }
                    catch { }
                }
                if (refocus && target != IntPtr.Zero && NativeMethods.IsWindow(target))
                {
                    NativeMethods.ForceForeground(target);
                }
            });
        }

        private void ShowBalloon(string title, string text, ToolTipIcon icon)
        {
            if (_tray == null || !_tray.Visible) return;
            if (string.IsNullOrEmpty(text)) return;
            if (text.Length > 240) text = text.Substring(0, 237) + "...";
            try
            {
                _tray.ShowBalloonTip(5000, title, text, icon);
            }
            catch
            {
            }
        }

        public void ExitApp()
        {
            if (_exiting) return;
            _exiting = true;
            Cleanup();
            ExitThread();
        }

        private void Cleanup()
        {
            if (_cleaned) return;
            _cleaned = true;

            if (_picker != null)
            {
                try
                {
                    _picker.DismissSilently();
                    _picker.Dispose();
                }
                catch
                {
                }
                _picker = null;
            }

            try { _hotkeys.UnregisterAll(); }
            catch { }

            try
            {
                _manager.PrepareExit();
                _manager.Hide();
            }
            catch { }

            if (_showWait != null)
            {
                try { _showWait.Unregister(null); }
                catch { }
                _showWait = null;
            }
            if (_showEvent != null)
            {
                try { _showEvent.Close(); }
                catch { }
                _showEvent = null;
            }

            try { _tray.Visible = false; }
            catch { }

            try { _hotkeys.Dispose(); } catch { }
            try { _paste.Dispose(); } catch { }
            try { _manager.Dispose(); } catch { }
            try { _menu.Dispose(); } catch { }
            try { _tray.Dispose(); } catch { }
            try { _activeIcon.Dispose(); } catch { }
            try { _pausedIcon.Dispose(); } catch { }
        }

        protected override void Dispose(bool disposing)
        {
            if (disposing)
            {
                _exiting = true;
                Cleanup();
            }
            base.Dispose(disposing);
        }
    }
}

// ----- 10_SelfTest.cs -----
namespace PasteNotes
{
    internal static class SelfTest
    {
        private static StringBuilder _log;
        private static int _failures;
        private static int _checks;

        public static string LogPath
        {
            get { return Path.Combine(Path.GetTempPath(), "PasteNotesSelfTest.log"); }
        }

        public static int Run()
        {
            _log = new StringBuilder();
            _failures = 0;
            _checks = 0;
            string temporary = null;

            WriteLine("PasteNotes self-test");
            try
            {
                TestDefaults();
                TestJson();
                TestStore(out temporary);
                TestHotkeys();
                TestRtl();
                TestPalette();
                TestIcons();
            }
            catch (Exception ex)
            {
                Fail("Unexpected self-test error: " + ex);
            }
            finally
            {
                if (!string.IsNullOrEmpty(temporary))
                {
                    try { Directory.Delete(temporary, true); }
                    catch { }
                }
            }

            string result = _failures == 0
                ? "PASS - " + _checks.ToString(CultureInfo.InvariantCulture) + " checks"
                : "FAIL - " + _failures.ToString(CultureInfo.InvariantCulture) +
                  " of " + _checks.ToString(CultureInfo.InvariantCulture) + " checks failed";
            WriteLine(result);

            try
            {
                File.WriteAllText(LogPath, _log.ToString(), new UTF8Encoding(false));
            }
            catch
            {
            }

            return _failures == 0 ? 0 : 1;
        }

        private static void TestDefaults()
        {
            AppData data = Defaults.CreateAppData();
            Check(data.Groups.Count == 8, "defaults contain 8 groups (found " + data.Groups.Count + ")");

            int templateCount = 0;
            HashSet<string> ids = new HashSet<string>(StringComparer.Ordinal);
            foreach (GroupData group in data.Groups)
            {
                templateCount += group.Templates.Count;
                Check(!string.IsNullOrEmpty(group.Id), "group id is set");
                Check(ids.Add(group.Id), "group id is unique");
                Check(!string.IsNullOrEmpty(group.Name), "group name is set");
                Check(!string.IsNullOrEmpty(group.Hotkey), "group hotkey is set");
                Check(Palette.Normalize(group.Color) == group.Color, "group color is normalized");

                try
                {
                    HotkeyParser.Parse(group.Hotkey);
                    Check(true, "group hotkey parses: " + group.Hotkey);
                }
                catch (HotkeyException ex)
                {
                    Fail("group hotkey does not parse (" + group.Hotkey + "): " + ex.Message);
                }

                foreach (TemplateData template in group.Templates)
                {
                    Check(!string.IsNullOrEmpty(template.Id), "template id is set");
                    Check(ids.Add(template.Id), "template id is unique");
                    Check(template.Title != null, "template title is not null");
                    Check(template.Content != null, "template content is not null");
                }
            }
            Check(templateCount == 78, "defaults contain 78 templates (found " + templateCount + ")");

            string greeting = Defaults.GreetingForSelfTest();
            Check(greeting.Length > 0, "expected greeting is embedded");
            Check(RtlHelper.IsRtl(greeting), "expected greeting is right-to-left");
            if (data.Groups.Count > 0 && data.Groups[0].Templates.Count > 0)
            {
                string first = data.Groups[0].Templates[0].Content;
                Check(first == greeting, "first default template matches the expected greeting");
                Check(RtlHelper.IsRtl(first), "first default template is right-to-left");
            }
            else
            {
                Fail("defaults do not contain the first template");
            }
        }

        private static void TestJson()
        {
            AppData data = Defaults.CreateAppData();
            data.Settings["restore_clipboard"] = false;
            string specialTitle = "Quote \" and <tag> & slash \\";
            string specialContent = "line 1\r\nline 2\ttabbed \u0001 control\r\nemoji \U0001F600";
            data.Groups[0].Templates[0].Title = specialTitle;
            data.Groups[0].Templates[0].Content = specialContent;

            string json = JsonCodec.SerializeAppData(data);
            Check(json.IndexOf("\"version\": 1", StringComparison.Ordinal) >= 0, "serialized JSON contains version");
            Check(json.IndexOf("\\u0001", StringComparison.Ordinal) >= 0, "serialized JSON escapes control characters");

            AppData parsed = JsonCodec.ParseAppData(json);
            Check(parsed.Groups.Count == data.Groups.Count, "JSON round-trip keeps group count");
            Check(parsed.Groups[0].Templates[0].Title == specialTitle, "JSON round-trip keeps special title");
            Check(parsed.Groups[0].Templates[0].Content == specialContent, "JSON round-trip keeps special content");
            object restoreValue;
            bool restore = parsed.Settings.TryGetValue("restore_clipboard", out restoreValue) &&
                           restoreValue != null &&
                           !Convert.ToBoolean(restoreValue, CultureInfo.InvariantCulture);
            Check(restore, "JSON round-trip keeps settings");

            bool versionRejected = false;
            try
            {
                JsonCodec.ParseAppData(json.Replace("\"version\": 1", "\"version\": 99"));
            }
            catch (DataVersionException)
            {
                versionRejected = true;
            }
            Check(versionRejected, "newer data version is rejected");

            bool rootRejected = false;
            try
            {
                JsonCodec.ParseAppData("[]");
            }
            catch (FormatException)
            {
                rootRejected = true;
            }
            Check(rootRejected, "non-object JSON root is rejected");
        }

        private static void TestStore(out string temporary)
        {
            temporary = Path.Combine(
                Path.GetTempPath(),
                "PasteNotesSelfTest-" + Guid.NewGuid().ToString("N"));
            Directory.CreateDirectory(temporary);
            string path = Path.Combine(temporary, "data.json");

            DataStore store = new DataStore(path);
            store.Load();
            Check(File.Exists(path), "store seeds a new data file");
            Check(store.Data.Groups.Count == 8, "seeded store has 8 groups");

            GroupData group = store.AddGroup("Self-test group");
            Check(group != null && group.Templates.Count == 0, "store adds a group");
            TemplateData template = store.AddTemplate(group);
            template.Title = "Self-test template";
            template.Content = "Hello from the self-test.";
            store.Save();

            store.SetSetting("restore_clipboard", false);
            store.Save();
            Check(File.Exists(path + ".bak"), "store writes a backup file");

            DataStore loaded = new DataStore(path);
            loaded.Load();
            Check(loaded.Data.Groups.Count == 9, "saved store reloads all groups");
            Check(!loaded.GetBoolSetting("restore_clipboard", true), "saved store reloads settings");
            GroupData loadedGroup = loaded.FindGroup(group.Id);
            Check(loadedGroup != null, "saved store keeps group ids");
            if (loadedGroup != null)
            {
                TemplateData loadedTemplate = loaded.FindTemplate(group.Id, template.Id);
                Check(loadedTemplate != null, "saved store keeps template ids");
                if (loadedTemplate != null)
                {
                    Check(loadedTemplate.Title == "Self-test template", "saved store keeps template titles");
                    Check(loadedTemplate.Content == "Hello from the self-test.", "saved store keeps template content");
                }
            }

            File.Delete(path + ".bak");
            File.WriteAllText(path, "this is not json", new UTF8Encoding(false));
            DataStore recovered = new DataStore(path);
            recovered.Load();
            Check(recovered.Data.Groups.Count == 8, "unreadable data is replaced with defaults");
            string[] quarantined = Directory.GetFiles(temporary, "data.corrupt-*.json");
            Check(quarantined.Length == 1, "unreadable data is quarantined");
        }

        private static void TestHotkeys()
        {
            Check(HotkeyParser.Normalize("ctrl+alt+1") == "Ctrl+Alt+1", "hotkey normalization");
            Check(HotkeyParser.Normalize("F9") == "F9", "function key hotkey is allowed");
            Check(HotkeyParser.Normalize("Ctrl++") == "Ctrl++", "plus key hotkey is allowed");
            Check(HotkeyParser.Normalize("Shift+Win+K") == "Shift+Win+K", "Windows key hotkey is allowed");

            CheckRejected("A");
            CheckRejected("Ctrl+Alt");
            CheckRejected("Ctrl+A+B");
            CheckRejected("Ctrl+NotAKey");

            Check(HotkeyParser.DisplayFromKey(Keys.A) == "A", "display name for letter key");
            Check(HotkeyParser.DisplayFromKey(Keys.NumPad5) == "Num5", "display name for numpad key");
            Check(HotkeyParser.DisplayFromKey(Keys.Oemtilde) == "`", "display name for punctuation key");
        }

        private static void CheckRejected(string text)
        {
            bool rejected = false;
            try
            {
                HotkeyParser.Parse(text);
            }
            catch (HotkeyException)
            {
                rejected = true;
            }
            Check(rejected, "invalid hotkey is rejected: " + text);
        }

        private static void TestRtl()
        {
            Check(RtlHelper.IsRtl("\u0634\u0643\u0631\u0627\u064B \u0644\u062A\u0648\u0627\u0635\u0644\u0643\u0645"),
                "Arabic text is detected as right-to-left");
            Check(RtlHelper.IsRtl("\u05E9\u05DC\u05D5\u05DD"), "Hebrew text is detected as right-to-left");
            Check(!RtlHelper.IsRtl("Hello world"), "English text is detected as left-to-right");
            Check(!RtlHelper.IsRtl("12345"), "digit-only text is neutral");
            Check(!RtlHelper.IsRtl(""), "empty text is neutral");
        }

        private static void TestPalette()
        {
            Check(Palette.Normalize("BLUE") == "blue", "palette keys are case-insensitive");
            Check(Palette.Normalize("not-a-color") == "yellow", "unknown colors fall back to yellow");
            Check(Palette.HexValue("blue") == "#56A3E8", "palette hex values are stable");
            Check(Palette.Keys().Length == 8, "palette exposes 8 colors");
        }

        private static void TestIcons()
        {
            using (System.Drawing.Icon active = IconFactory.Create(false))
            using (System.Drawing.Icon paused = IconFactory.Create(true))
            {
                Check(active != null && active.Width >= 16, "active tray icon is generated");
                Check(paused != null && paused.Width >= 16, "paused tray icon is generated");
            }
        }

        private static void Check(bool condition, string message)
        {
            _checks++;
            if (condition)
            {
                WriteLine("ok   " + message);
                return;
            }
            _failures++;
            WriteLine("FAIL " + message);
        }

        private static void Fail(string message)
        {
            _checks++;
            _failures++;
            WriteLine("FAIL " + message);
        }

        private static void WriteLine(string message)
        {
            _log.AppendLine(message);
        }
    }
